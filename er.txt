@Composable
fun Desktop(
    wallpaper: WallpaperState,
    desktopItems: List<DesktopItem>,
    desktopShortcuts: List<DesktopShortcut>,
    isDark: Boolean,
    viewModel: LauncherViewModel,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val uiState by viewModel.uiState.collectAsState()

    Box(modifier = modifier.fillMaxSize()) {
        // ── Wallpaper background ──
        if (wallpaper.homeWallpaperUri.isNotEmpty()) {
            AsyncImage(
                model = Uri.parse(wallpaper.homeWallpaperUri),
                contentDescription = null,
                modifier = Modifier.fillMaxSize(),
                contentScale = ContentScale.Crop
            )
        } else {
            val gradient = wallpaperGradients[wallpaper.homeWallpaperIndex % wallpaperGradients.size]
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(
                        Brush.linearGradient(gradient, start = Offset(0f, 0f), end = Offset(2000f, 1200f))
                    )
            )
            Canvas(modifier = Modifier.fillMaxSize()) {
                drawCircle(
                    brush = Brush.radialGradient(
                        colors = listOf(Color(0x30FFFFFF), Color(0x00FFFFFF)),
                        center = Offset(size.width * 0.6f, size.height * 0.4f),
                        radius = size.width * 0.5f
                    ),
                    radius = size.width * 0.5f,
                    center = Offset(size.width * 0.6f, size.height * 0.4f)
                )
            }
        }

        // ── Long-press for context menu ──
        Box(
            modifier = Modifier
                .fillMaxSize()
                .pointerInput(Unit) {
                    detectTapGestures(
                        onLongPress = { offset ->
                            viewModel.openDesktopContextMenu(offset.x, offset.y)
                        },
                        onDoubleTap = {
                            viewModel.toggleTaskbar()
                        },
                        onTap = {
                            if (uiState.isDesktopContextMenuOpen) viewModel.closeDesktopContextMenu()
                            else viewModel.dismissAllOverlays()
                        }
                    )
                }
        )

        // ── Desktop icons grid (right side) ──
        Column(
            modifier = Modifier
                .align(Alignment.TopEnd)
                .padding(top = 16.dp, end = 16.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            // System items
            desktopItems.forEach { item ->
                SystemDesktopIcon(
                    item = item,
                    recycleBinCount = uiState.recycleBinItems.size,
                    onClick = {
                        when (item.type) {
                            DesktopItemType.THIS_PC, DesktopItemType.NETWORK ->
                                viewModel.openWindow(LauncherScreen.FILE_EXPLORER)
                            DesktopItemType.RECYCLE_BIN ->
                                viewModel.openWindow(LauncherScreen.RECYCLE_BIN)
                            DesktopItemType.SETTINGS_ICON ->
                                viewModel.openWindow(LauncherScreen.SETTINGS)
                            else -> {}
                        }
                    }
                )
            }

            // User shortcuts
            desktopShortcuts.forEach { shortcut ->
                DesktopShortcutIcon(
                    shortcut = shortcut,
                    onClick = {
                        when (shortcut.type) {
                            "app" -> {
                                val app = uiState.installedApps.find { it.packageName == shortcut.packageName }
                                    ?: AppInfo(shortcut.label, shortcut.packageName, null)
                                viewModel.openApp(context, app)
                            }
                            "file", "folder" -> viewModel.openFileWithSystem(context, shortcut.targetPath)
                            else -> {}
                        }
                    },
                    onRemove = { viewModel.removeDesktopShortcut(shortcut.id) }
                )
            }
        }

        // ── Desktop Context Menu ──
        if (uiState.isDesktopContextMenuOpen) {
            DesktopContextMenu(
                x = uiState.desktopContextMenuX,
                y = uiState.desktopContextMenuY,
                isDark = isDark,
                onDismiss = { viewModel.closeDesktopContextMenu() },
                onNewFolder = { name ->
                    viewModel.closeDesktopContextMenu()
                    viewModel.createDesktopFolder(name)
                },
                onNewTextFile = { name ->
                    viewModel.closeDesktopContextMenu()
                    viewModel.createDesktopTextFile(name)
                },
                onRefresh = { viewModel.closeDesktopContextMenu() },
                onPersonalize = {
                    viewModel.closeDesktopContextMenu()
                    viewModel.openWallpaperPicker(WallpaperTarget.HOME)
                },
                onDisplaySettings = {
                    viewModel.closeDesktopContextMenu()
                    viewModel.openWindow(LauncherScreen.SETTINGS)
                }
            )
        }
    }
}

@Composable
fun SystemDesktopIcon(
    item: DesktopItem,
    recycleBinCount: Int,
    onClick: () -> Unit
) {
    var selected by remember { mutableStateOf(false) }
    val icon = when (item.type) {
        DesktopItemType.THIS_PC -> Icons.Default.Computer
        DesktopItemType.RECYCLE_BIN -> if (recycleBinCount > 0) Icons.Default.Delete else Icons.Default.DeleteOutline
        DesktopItemType.SETTINGS_ICON -> Icons.Default.Settings
        DesktopItemType.NETWORK -> Icons.Default.Wifi
        else -> Icons.Default.Folder
    }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        modifier = Modifier
            .width(80.dp)
            .clip(RoundedCornerShape(6.dp))
            .background(if (selected) Color(0x30FFFFFF) else Color.Transparent)
            .pointerInput(Unit) {
                detectTapGestures(
                    onPress = { selected = true; tryAwaitRelease(); selected = false },
                    onTap = { onClick() }
                )
            }
            .padding(6.dp)
    ) {
        Box(
            modifier = Modifier.size(44.dp).background(Win11Colors.AccentBlue.copy(alpha = 0.8f), RoundedCornerShape(8.dp)),
            contentAlignment = Alignment.Center
        ) {
            Icon(icon, null, tint = Color.White, modifier = Modifier.size(26.dp))
            if (item.type == DesktopItemType.RECYCLE_BIN && recycleBinCount > 0) {
                Badge(
                    modifier = Modifier.align(Alignment.TopEnd).offset(4.dp, (-4).dp),
                    containerColor = Win11Colors.DangerRed
                ) { Text("$recycleBinCount", fontSize = 9.sp) }
            }
        }
        Spacer(Modifier.height(4.dp))
        Text(
            text = item.label,
            style = MaterialTheme.typography.labelSmall,
            color = Color.White,
            textAlign = TextAlign.Center,
            maxLines = 2,
            overflow = TextOverflow.Ellipsis,
            fontSize = 10.sp
        )
    }
}

@Composable
fun DesktopShortcutIcon(
    shortcut: DesktopShortcut,
    onClick: () -> Unit,
    onRemove: () -> Unit
) {
    var selected by remember { mutableStateOf(false) }
    var showContextMenu by remember { mutableStateOf(false) }

    val icon = when (shortcut.type) {
        "folder" -> Icons.Default.Folder
        "file" -> Icons.Default.InsertDriveFile
        else -> Icons.Default.Apps
    }

    Box {
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
            modifier = Modifier
                .width(80.dp)
                .clip(RoundedCornerShape(6.dp))
                .background(if (selected) Color(0x30FFFFFF) else Color.Transparent)
                .pointerInput(Unit) {
                    detectTapGestures(
                        onPress = { selected = true; tryAwaitRelease(); selected = false },
                        onTap = { onClick() },
                        onLongPress = { showContextMenu = true }
                    )
                }
                .padding(6.dp)
        ) {
            Box(
                modifier = Modifier.size(44.dp),
                contentAlignment = Alignment.Center
            ) {
                Box(
                    modifier = Modifier.fillMaxSize().background(iconTint.copy(alpha = 0.15f), RoundedCornerShape(8.dp)),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(icon, null, tint = iconTint, modifier = Modifier.size(28.dp))
                }
                // Shortcut arrow
                Box(
                    modifier = Modifier.align(Alignment.BottomStart).size(14.dp)
                        .background(Color(0xFF333333), RoundedCornerShape(3.dp)),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(Icons.Default.Launch, null, tint = Color.White, modifier = Modifier.size(10.dp))
                }
            }
            Spacer(Modifier.height(4.dp))
            Text(shortcut.label, style = MaterialTheme.typography.labelSmall, color = Color.White, textAlign = TextAlign.Center, maxLines = 2, overflow = TextOverflow.Ellipsis, fontSize = 10.sp)
        }

        if (showContextMenu) {
            DropdownMenu(
                expanded = true,
                onDismissRequest = { showContextMenu = false },
                modifier = Modifier.background(Win11Colors.ContextMenuBg).width(180.dp)
            ) {
                DropdownMenuItem(
                    text = { Text("Open", color = Color.White, fontSize = 13.sp) },
                    onClick = { showContextMenu = false; onClick() },
                    leadingIcon = { Icon(Icons.Default.OpenInNew, null, tint = Color.White, modifier = Modifier.size(16.dp)) }
                )
                Divider(color = Color.White.copy(alpha = 0.1f))
                DropdownMenuItem(
                    text = { Text("Remove shortcut", color = Color.White, fontSize = 13.sp) },
                    onClick = { showContextMenu = false; onRemove() },
                    leadingIcon = { Icon(Icons.Default.Delete, null, tint = Win11Colors.DangerRed, modifier = Modifier.size(16.dp)) }
                )
            }
        }
    }
}

@Composable
fun DesktopContextMenu(
    x: Float, y: Float,
    isDark: Boolean,
    onDismiss: () -> Unit,
    onNewFolder: (String) -> Unit,
    onNewTextFile: (String) -> Unit,
    onRefresh: () -> Unit,
    onPersonalize: () -> Unit,
    onDisplaySettings: () -> Unit
) {
    var showNewSubMenu by remember { mutableStateOf(false) }
    var showNewFolderDialog by remember { mutableStateOf(false) }
    var showNewFileDialog by remember { mutableStateOf(false) }
    var newItemName by remember { mutableStateOf("") }

    val menuBg = if (isDark) Win11Colors.ContextMenuBg else Win11Colors.ContextMenuBgLight
    val menuText = if (isDark) Color.White else Win11Colors.TextPrimaryLight

    Box(
        modifier = Modifier
            .fillMaxSize()
            .pointerInput(Unit) { detectTapGestures(onTap = { onDismiss() }) }
    ) {
        Card(
            modifier = Modifier
                .offset { IntOffset(x.toInt().coerceIn(0, 900), y.toInt().coerceIn(0, 400)) }
                .width(220.dp),
            shape = RoundedCornerShape(8.dp),
            colors = CardDefaults.cardColors(containerColor = menuBg),
            elevation = CardDefaults.cardElevation(8.dp)
        ) {
            Column(modifier = Modifier.padding(4.dp)) {
                ContextMenuItem(Icons.Default.Refresh, "Refresh", menuText) { onRefresh() }
                Divider(Modifier.padding(vertical = 2.dp), color = Color.White.copy(alpha = 0.1f))

                // View submenu placeholder
                ContextMenuItem(Icons.Default.ViewModule, "View", menuText, hasSubmenu = true) {}
                ContextMenuItem(Icons.Default.SortByAlpha, "Sort by", menuText, hasSubmenu = true) {}

                Divider(Modifier.padding(vertical = 2.dp), color = Color.White.copy(alpha = 0.1f))

                // New submenu
                Box {
                    ContextMenuItem(Icons.Default.Add, "New", menuText, hasSubmenu = true) {
                        showNewSubMenu = !showNewSubMenu
                    }
                    DropdownMenu(
                        expanded = showNewSubMenu,
                        onDismissRequest = { showNewSubMenu = false },
                        modifier = Modifier.background(menuBg).width(160.dp)
                    ) {
                        DropdownMenuItem(
                            text = { Text("Folder", color = menuText, fontSize = 13.sp) },
                            onClick = { showNewSubMenu = false; newItemName = "New Folder"; showNewFolderDialog = true },
                            leadingIcon = { Icon(Icons.Default.Folder, null, tint = Color(0xFFFFC107), modifier = Modifier.size(16.dp)) }
                        )
                        DropdownMenuItem(
                            text = { Text("Text File", color = menuText, fontSize = 13.sp) },
                            onClick = { showNewSubMenu = false; newItemName = "New Text File"; showNewFileDialog = true },
                            leadingIcon = { Icon(Icons.Default.Description, null, tint = Color.White, modifier = Modifier.size(16.dp)) }
                        )
                    }
                }

                Divider(Modifier.padding(vertical = 2.dp), color = Color.White.copy(alpha = 0.1f))
                ContextMenuItem(Icons.Default.Palette, "Personalize", menuText) { onPersonalize() }
                ContextMenuItem(Icons.Default.Monitor, "Display settings", menuText) { onDisplaySettings() }
            }
        }
    }

    if (showNewFolderDialog) {
        NewItemDialog(
            title = "New Folder",
            label = "Folder name",
            initialValue = "New Folder",
            onConfirm = { name -> showNewFolderDialog = false; onNewFolder(name) },
            onDismiss = { showNewFolderDialog = false }
        )
    }

    if (showNewFileDialog) {
        NewItemDialog(
            title = "New Text File",
            label = "File name",
            initialValue = "New Text File",
            onConfirm = { name -> showNewFileDialog = false; onNewTextFile(name) },
            onDismiss = { showNewFileDialog = false }
        )
    }
}

@Composable
private fun ContextMenuItem(
    icon: androidx.compose.ui.graphics.vector.ImageVector,
    label: String,
    textColor: Color,
    hasSubmenu: Boolean = false,
    onClick: () -> Unit
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(4.dp))
            .pointerInput(Unit) { detectTapGestures(onTap = { onClick() }) }
            .padding(horizontal = 12.dp, vertical = 8.dp),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(10.dp)
    ) {
        Icon(icon, null, tint = textColor.copy(alpha = 0.8f), modifier = Modifier.size(15.dp))
        Text(label, style = MaterialTheme.typography.bodySmall, color = textColor, modifier = Modifier.weight(1f))
        if (hasSubmenu) Icon(Icons.Default.ChevronRight, null, tint = textColor.copy(alpha = 0.5f), modifier = Modifier.size(14.dp))
    }
}

@Composable
fun NewItemDialog(
    title: String,
    label: String,
    initialValue: String,
    onConfirm: (String) -> Unit,
    onDismiss: () -> Unit
) {
    var text by remember { mutableStateOf(initialValue) }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(title) },
        text = {
            OutlinedTextField(
                value = text,
                onValueChange = { text = it },
                label = { Text(label) },
                singleLine = true,
                modifier = Modifier.fillMaxWidth()
            )
        },
        confirmButton = {
            TextButton(onClick = { if (text.isNotBlank()) onConfirm(text.trim()) }) { Text("Create") }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) { Text("Cancel") }
        }
    )
}
